package cl.oyl.mobile.viewmodel.externa

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import cl.oyl.mobile.data.remote.ExternalFood
import cl.oyl.mobile.data.remote.RemoteApis
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class ExternalApiViewModel(app: Application) : AndroidViewModel(app) {
    private val _foods = MutableStateFlow<List<ExternalFood>>(emptyList())
    val foods = _foods.asStateFlow()
    private val _loading = MutableStateFlow(false)
    val loading = _loading.asStateFlow()
    private val _error = MutableStateFlow<String?>(null)
    val error = _error.asStateFlow()

    // Consulta en segundo plano y expone carga, resultados o error a Compose.
    fun search() {
        if (_loading.value) return
        viewModelScope.launch(Dispatchers.IO) {
            _loading.value = true
            _error.value = null
            try {
                val result = RemoteApis.foodFacts.searchFoods().products
                    .filter { !it.name.isNullOrBlank() }
                    .distinctBy { it.name?.lowercase() }
                _foods.value = result
                if (result.isEmpty()) _error.value = "La API no devolvió productos para esta búsqueda."
            } catch (e: Exception) {
                if (e is CancellationException) throw e
                _error.value = "No se pudo consultar Open Food Facts. Revisa la conexión e inténtalo de nuevo."
            } finally {
                _loading.value = false
            }
        }
    }
}
